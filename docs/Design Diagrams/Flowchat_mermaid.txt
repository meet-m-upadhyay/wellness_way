flowchart TD
    %% Nodes
    User(("User\n(Frontend)"))
    FastAPI["FastAPI Endpoints\n(api/endpoints/diet_plans_ml.py)"]
    DB[("PostgreSQL Database\n(Supabase / Local)")]
    Orch{"ML Orchestrator"}
    Disc["Discovery Engine\n(FAISS / Canonicalizer)"]
    Assem["Daily Assembler"]
    GenAI["GenAI Service"]
    LLM(("External LLM\n(Groq/OpenAI)"))
    Val["Validation Engine"]
    Nut["Nutrition Engine"]

    %% User Interaction
    User -- "POST /api/diet-plans-ml/daily" --> FastAPI
    
    %% API to Database
    FastAPI -- "Query HealthContextDocument" --> DB
    DB -- "Returns HCD Context JSON\n(Macros, Allergies, Diet Type)" --> FastAPI
    
    %% Getting to the Orchestrator
    FastAPI -- "generate_plan(context)" --> Orch
    Orch -- "_extract_constraints(macros, allergies)" --> Orch
    
    %% Discovery Phase
    Orch -- "discover_daily_portfolio()" --> Disc
    Disc -- "Query vector embeddings" --> DB
    DB -- "Returns filtered ingredients" --> Disc
    Disc -- "Returns Portfolio List" --> Orch
    
    %% Assembly Phase
    Orch -- "assemble_day(portfolio, targets)" --> Assem
    Assem -- "Returns mathematically distributed Raw Meals" --> Orch
    
    %% GenAI Loop
    Orch -- "Loop: generate_meal_text(ingredients, diet)" --> GenAI
    GenAI -- "REST API Call\n(Prompt + JSON schema)" --> LLM
    LLM -- "Returns JSON (meal_name, steps)" --> GenAI
    GenAI -- "Returns parsed GenAIMealText" --> Orch
    
    %% Validation & Nutrition
    Orch -- "validate_daily_plan(meals, targets)" --> Val
    Val -- "Returns Validation Warnings" --> Orch
    Orch -- "calculate_daily_nutrition(meals)" --> Nut
    Nut -- "Returns Daily Totals (Rollup)" --> Orch
    
    %% Return Path
    Orch -- "Returns Full Plan Dict" --> FastAPI
    FastAPI -- "CREATE DietPlan Record" --> DB
    DB -- "Returns Saved DietPlan" --> FastAPI
    FastAPI -- "HTTP 201 Created\n(JSON Response)" --> User
